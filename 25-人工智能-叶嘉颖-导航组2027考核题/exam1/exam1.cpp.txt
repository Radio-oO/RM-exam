#include <iostream>

using namespace std;

int main()
{
    int arr[]={5,8,2,10,7,15,13,11,20,18};
    int t,i,j;
    int n=sizeof(arr)/sizeof(arr[0]);
    cout<<"排序前: ";
    for (i=0;i<n;i++)
    {
        cout<<arr[i]<<" ";
    }
    cout<<endl;
    for(i=0;i<n-1;i++)   //i<n-1,表示循环轮数
    {
        for(j=0;j<n-1-i;j++) //j为每轮比较次数
        {
            if(arr[j]>arr[j+1])
            {
               t=arr[j];
               arr[j]=arr[j+1];
               arr[j+1]=t;
            }

        }
    }

   cout<<"排序后: ";
    for (i=0;i<n;i++)
    {
        cout<<arr[i]<<" ";
    }
    cout<<endl;
    return 0;
}
